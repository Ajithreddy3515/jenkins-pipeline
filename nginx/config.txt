# HTTPS (PORT 443)

server {
    listen 443 ssl;
    listen [::]:443 ssl;

    server_name test.pmrcoretech.com;

     root /var/www/html/Website;
     index index.html;

     ssl_certificate /etc/letsencrypt/live/test.pmrcoretech.com/fullchain.pem;
     ssl_certificate_key /etc/letsencrypt/live/test.pmrcoretech.com/privkey.pem;

    # ===========================
    # SECURITY HEADERS
    # ===========================

    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "0" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;

    add_header Content-Security-Policy "
    default-src 'self';
    script-src 'self' 'unsafe-inline';
    style-src 'self' 'unsafe-inline' https:;
    img-src 'self' data: https:;
    connect-src 'self' https: wss:;
    font-src 'self' data: https:;
    " always;

    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains; preload" always;

#global timeouts
    client_header_timeout 15s;
    client_body_timeout 15s;

    keepalive_timeout 30s;
 
    send_timeout 30s;


# proxy time outs for backedn is stoped nginx should close the connection
    proxy_connect_timeout 5s;
 
    proxy_send_timeout 30s;

    proxy_read_timeout 60s;


# for uploading the files max size limit
    client_max_body_size 10M;

# prevention for the env, git , paawd etc
    location ~ /\. {
    deny all;
}

# preventing the backup files from attackers to download 
    location ~* \.(bak|old|zip|sql|tar|gz)$ {
    deny all;
}

# buffering when backend sends data but user network slows then backend os opened insted we store in nginx for better performance
    proxy_buffering on;

    proxy_buffer_size 16k;

    proxy_buffers 8 16k;

# for accesing the things from user with their permisison 
    add_header Permissions-Policy "camera=(), microphone=(), geolocation=()" always;

# allowing only requried api's 
    if ($request_method !~ ^(GET|POST|PUT|DELETE|PATCH|OPTIONS)$) {
    return 405;
}

##################################################################################
# Cache Vite assets
    location /assets/ {
        expires 30d;
        add_header Cache-Control "public, immutable";
    }

    # Cache images/icons/fonts
    location ~* \.(png|jpg|jpeg|gif|svg|ico|woff2)$ {
        expires 30d;
        add_header Cache-Control "public";
    }

    # Don't cache index.html
    location = /index.html {
        expires -1;
        add_header Cache-Control "no-cache";
    }
####################################################################################

    
    # ========= WEBSITE =========
    location / {

        try_files $uri $uri/ /index.html;

        proxy_set_header X-Forwarded-Proto https;
        proxy_set_header Cookie $http_cookie;
        proxy_set_header Authorization $http_authorization;

        proxy_next_upstream error timeout http_500 http_502 http_503 http_504;
