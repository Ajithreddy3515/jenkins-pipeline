# ============================================================
# NGINX REVERSE PROXY + SSL CONFIGURATION
# ============================================================
# This is a sanitized example configuration.
# Production domains, certificates, paths and secrets are
# intentionally replaced with placeholders.
# ============================================================


# ============================================================
# HTTP -> HTTPS REDIRECT
# ============================================================

server {
    listen 80;
    listen [::]:80;

    server_name example.com;

    return 301 https://$host$request_uri;
}


# ============================================================
# LOCAL BACKEND PROXY
# ============================================================

server {
    listen 127.0.0.1:8090;

    server_name localhost;

    location / {
        proxy_pass http://motor_backend;

        proxy_http_version 1.1;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}


# ============================================================
# UPSTREAM CONFIGURATION
# ============================================================

include /etc/nginx/upstreams/*.conf;


# ============================================================
# HTTPS SERVER
# ============================================================

server {

    listen 443 ssl;
    listen [::]:443 ssl;

    server_name example.com;


    # ========================================================
    # FRONTEND
    # ========================================================

    root /var/www/html/Website;
    index index.html;


    # ========================================================
    # SSL / TLS
    # ========================================================

    ssl_certificate
        /etc/letsencrypt/live/example.com/fullchain.pem;

    ssl_certificate_key
        /etc/letsencrypt/live/example.com/privkey.pem;


    # ========================================================
    # SECURITY HEADERS
    # ========================================================

    add_header X-Frame-Options "SAMEORIGIN" always;

    add_header X-Content-Type-Options "nosniff" always;

    add_header Referrer-Policy
        "strict-origin-when-cross-origin" always;

    add_header Permissions-Policy
        "camera=(), microphone=(), geolocation=()" always;

    add_header Strict-Transport-Security
        "max-age=31536000; includeSubDomains; preload"
        always;


    # ========================================================
    # CONTENT SECURITY POLICY
    # ========================================================

    add_header Content-Security-Policy "
        default-src 'self';
        script-src 'self' 'unsafe-inline';
        style-src 'self' 'unsafe-inline' https:;
        img-src 'self' data: https:;
        connect-src 'self' https: wss:;
        font-src 'self' data: https:;
    " always;


    # ========================================================
    # TIMEOUTS
    # ========================================================

    client_header_timeout 15s;
    client_body_timeout 15s;

    keepalive_timeout 30s;

    send_timeout 30s;

    proxy_connect_timeout 5s;
    proxy_send_timeout 30s;
    proxy_read_timeout 60s;


    # ========================================================
    # REQUEST SIZE
    # ========================================================

    client_max_body_size 10M;


    # ========================================================
    # BLOCK HIDDEN FILES
    # ========================================================

    location ~ /\. {
        deny all;
    }


    # ========================================================
    # BLOCK BACKUP / DATABASE FILES
    # ========================================================

    location ~* \.(bak|old|zip|sql|tar|gz)$ {
        deny all;
    }


    # ========================================================
    # PROXY BUFFERING
    # ========================================================

    proxy_buffering on;

    proxy_buffer_size 16k;

    proxy_buffers 8 16k;


    # ========================================================
    # ALLOWED HTTP METHODS
    # ========================================================

    if ($request_method !~ ^(GET|POST|PUT|DELETE|PATCH|OPTIONS)$) {
        return 405;
    }


    # ========================================================
    # STATIC ASSET CACHING
    # ========================================================

    location /assets/ {
        expires 30d;
        add_header Cache-Control "public, immutable";
    }


    location ~* \.(png|jpg|jpeg|gif|svg|ico|woff2)$ {
        expires 30d;
        add_header Cache-Control "public";
    }


    # ========================================================
    # DO NOT CACHE INDEX.HTML
    # ========================================================

    location = /index.html {
        expires -1;
        add_header Cache-Control "no-cache";
    }


    # ========================================================
    # MAIN WEBSITE
    # ========================================================

    location / {
        try_files $uri $uri/ /index.html;
    }


    # ========================================================
    # ADMIN FRONTEND
    # ========================================================

    location = /admin {
        return 302 /admin/$is_args$args;
    }

    location ^~ /admin/ {

        alias /var/www/html/Admin/;

        index index.html;

        try_files $uri $uri/ /admin/index.html;
    }


    # ========================================================
    # ADMIN BACKEND
    # ========================================================

    location /admin-api/ {

        proxy_pass http://admin_backend/;

        proxy_http_version 1.1;

        proxy_set_header Host $host;

        proxy_set_header X-Real-IP $remote_addr;

        proxy_set_header X-Forwarded-For
            $proxy_add_x_forwarded_for;

        proxy_set_header X-Forwarded-Proto https;

        proxy_set_header Cookie $http_cookie;

        proxy_set_header Authorization $http_authorization;


        # Performance

        proxy_buffering on;

        proxy_buffer_size 16k;

        proxy_buffers 8 16k;

        proxy_busy_buffers_size 32k;


        # Failover

        proxy_connect_timeout 2s;

        proxy_next_upstream
            error timeout
            http_500 http_502 http_503 http_504;
    }


    # ========================================================
    # MOTOR BACKEND
    # ========================================================

    location /motor-api/ {

        proxy_pass http://motor_backend/;

        proxy_http_version 1.1;

        proxy_set_header Host $host;

        proxy_set_header X-Real-IP $remote_addr;

        proxy_set_header X-Forwarded-For
            $proxy_add_x_forwarded_for;

        proxy_set_header X-Forwarded-Proto https;

        proxy_set_header Authorization $http_authorization;

        proxy_set_header X-SECRET-KEY $http_x_secret_key;


        # Performance

        proxy_buffering on;

        proxy_buffer_size 16k;

        proxy_buffers 8 16k;

        proxy_busy_buffers_size 32k;


        # Failover

        proxy_connect_timeout 2s;

        proxy_next_upstream
            error timeout
            http_500 http_502 http_503 http_504;
    }


    # ========================================================
    # ADMIN OTA WEBSOCKET
    # ========================================================

    location /ws/ota-status {

        proxy_pass http://admin_backend/ws/ota-status;

        proxy_http_version 1.1;

        proxy_set_header Upgrade $http_upgrade;

        proxy_set_header Connection "upgrade";

        proxy_set_header Host $host;

        proxy_set_header X-Real-IP $remote_addr;

        proxy_set_header X-Forwarded-For
            $proxy_add_x_forwarded_for;

        proxy_set_header X-Forwarded-Proto https;

        proxy_buffering off;
    }


    # ========================================================
    # MOTOR WEBSOCKET
    # ========================================================

    location /motor-websocket {

        proxy_pass http://motor_backend/motor-websocket;

        proxy_http_version 1.1;

        proxy_set_header Upgrade $http_upgrade;

        proxy_set_header Connection "upgrade";

        proxy_set_header Host $host;

        proxy_set_header X-Real-IP $remote_addr;

        proxy_set_header X-Forwarded-For
            $proxy_add_x_forwarded_for;

        proxy_set_header X-Forwarded-Proto https;

        proxy_set_header Authorization $http_authorization;

        proxy_connect_timeout 2s;

        proxy_next_upstream
            error timeout
            http_500 http_502 http_503 http_504;

        proxy_buffering off;
    }


    # ========================================================
    # SUPPORT ADMIN FRONTEND
    # ========================================================

    location = /support-admin {
        return 302 /support-admin/$is_args$args;
    }


    location /support-admin/ {

        alias /var/www/html/SupportAdmin/;

        index index.html;

        try_files
            $uri
            $uri/
            /support-admin/index.html;
    }


    # ========================================================
    # SUPPORT ADMIN API
    # ========================================================

    location /support-admin-api/ {

        proxy_pass http://127.0.0.1:8082/api/;

        proxy_http_version 1.1;

        proxy_set_header Host $host;

        proxy_set_header X-Real-IP $remote_addr;

        proxy_set_header X-Forwarded-For
            $proxy_add_x_forwarded_for;

        proxy_set_header X-Forwarded-Proto https;

        proxy_set_header Cookie $http_cookie;

        proxy_set_header Authorization $http_authorization;


        # Performance

        proxy_buffering on;

        proxy_buffer_size 16k;

        proxy_buffers 8 16k;

        proxy_busy_buffers_size 32k;
    }
}
